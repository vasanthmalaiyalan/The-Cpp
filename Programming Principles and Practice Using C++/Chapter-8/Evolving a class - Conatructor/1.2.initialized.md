```cpp
#include <iostream>

// ─── 1. Aggregate struct ───────────────────────────────────────────
struct Date
{
    int day;
    int month;
    int year;
};

// ─── 2. Partial init என்ன ஆகுது? ─────────────────────────────────
struct Point
{
    int x;
    int y;
    int z;
};

int main()
{
    // ── முழு initialization ──────────────────────────────────────
    Date d1 {23, 5, 2026};
    std::cout << "d1: " << d1.day << "/" << d1.month << "/" << d1.year << "\n";
    // Output: 23/5/2026

    // ── Copy-list-initialization (= உடன்) ── same result ─────────
    Date d2 = {1, 1, 2000};
    std::cout << "d2: " << d2.day << "/" << d2.month << "/" << d2.year << "\n";
    // Output: 1/1/2000

    // ── Empty brace → எல்லாமே 0 ──────────────────────────────────
    Date d3 {};
    std::cout << "d3: " << d3.day << "/" << d3.month << "/" << d3.year << "\n";
    // Output: 0/0/0

    // ── Partial init → மீதி members zero ─────────────────────────
    Point p1 {10, 20};          // z தரவில்லை → z = 0
    std::cout << "p1: " << p1.x << " " << p1.y << " " << p1.z << "\n";
    // Output: 10 20 0

    // ── Assignment vs Initialization ──────────────────────────────
    Date d4;                    // ⚠️  uninitialized! garbage values
    d4.day   = 15;
    d4.month = 8;
    d4.year  = 1947;
    std::cout << "d4: " << d4.day << "/" << d4.month << "/" << d4.year << "\n";
    // Output: 15/8/1947

    // ── Member-க்கு directly access செய்து மாற்றலாம் ─────────────
    Date d5 {10, 10, 2010};
    d5.year = 2025;             // ஒரு member மட்டும் மாற்றுவது
    std::cout << "d5: " << d5.day << "/" << d5.month << "/" << d5.year << "\n";
    // Output: 10/10/2025

    // ── Struct copy ───────────────────────────────────────────────
    Date d6 = d5;               // d5-ன் எல்லா members-உம் copy ஆகும்
    std::cout << "d6: " << d6.day << "/" << d6.month << "/" << d6.year << "\n";
    // Output: 10/10/2025

    // ── இவை compile error ஆகும் (comment-ல் மட்டும் காட்டுகிறோம்) ─
    // Date bad1 {23, 5, 2026, 99};  // ❌ members அதிகம்
    // Date bad2(23, 5, 2026);       // ❌ constructor இல்லை

    return 0;
}
```

### Output சுருக்கம்:
```
d1: 23/5/2026
d2: 1/1/2000
d3: 0/0/0
p1: 10 20 0
d4: 15/8/1947
d5: 10/10/2025
d6: 10/10/2025
```

### Key Points:

| Case | என்ன ஆகும் |
|---|---|
| `Date d {}` | எல்லாமே `0` |
| `Date d {23, 5}` | மீதி `0` |
| `Date d;` | ⚠️ Garbage values |
| `Date d = d2` | Full copy |

`Date d;` மட்டும் கவனமாக இருங்கள் — initialize செய்யாவிட்டால் **garbage values** வரும்!
