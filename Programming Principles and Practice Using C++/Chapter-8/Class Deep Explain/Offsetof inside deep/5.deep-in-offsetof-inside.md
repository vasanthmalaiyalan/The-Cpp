# Symbol Table concept புரிய — Best Code! 🎯

```cpp
#include <iostream>
#include <cstddef>
using namespace std;

// ======================================
// Compiler symbol table-ல் இருப்பது
// ======================================

struct Employee
{
    char   id;
    int    age;
    double salary;
};


int main()
{
    // ======================================
    // PART 1: Real address — RAM போகிறோம்!
    // ======================================

    Employee emp;
    emp.id     = 'A';
    emp.age    = 25;
    emp.salary = 50000.0;

    cout << "=== PART 1: Real Address (RAM access) ===\n\n";

    // base address
    void* base = &emp;
    cout << "Base address of emp : " << base << "\n";

    // age-இன் real address
    void* real_age_addr = &emp.age;
    cout << "Real address of age : " << real_age_addr << "\n";

    // base + offset = real address
    size_t base_num     = (size_t)base;
    size_t age_offset   = offsetof(Employee, age);
    size_t calculated   = base_num + age_offset;

    cout << "Base(" << base_num << ") + "
         << "Offset(" << age_offset << ") = "
         << calculated << "\n";

    // RAM-ல் போய் value எடுக்கிறோம்!
    cout << "Value inside RAM    : " << emp.age << "\n";


    // ======================================
    // PART 2: Fake address — RAM போகவில்லை!
    // ======================================

    cout << "\n=== PART 2: Fake Address (NO RAM access) ===\n\n";

    // 0 என்பது fake address
    // compiler symbol table மட்டும் பார்க்கிறது!
    void* fake_age_addr = &((Employee*)0)->age;

    cout << "Fake base address   : " << (void*)0 << "\n";
    cout << "Fake age address    : " << fake_age_addr << "\n";
    cout << "0 + offset(4)       : "
         << (size_t)fake_age_addr << "\n";

    // RAM போகவில்லை — கணக்கு மட்டும்!
    // *((Employee*)0) → இதை செய்தால் 💥 crash!


    // ======================================
    // PART 3: Symbol table values
    // ======================================

    cout << "\n=== PART 3: Compiler Symbol Table ===\n\n";

    cout << "id     offset : " << offsetof(Employee, id)     << "\n";
    cout << "age    offset : " << offsetof(Employee, age)    << "\n";
    cout << "salary offset : " << offsetof(Employee, salary) << "\n";
    cout << "struct size   : " << sizeof(Employee)           << "\n";

    // இந்த values compile time-லேயே தெரியும்!
    // எந்த emp object-உம் தேவையில்லை!


    // ======================================
    // PART 4: Real vs Fake — side by side
    // ======================================

    cout << "\n=== PART 4: Real vs Fake Comparison ===\n\n";

    // Real:
    size_t real  = (size_t)&emp.age - (size_t)&emp;
    // Fake:
    size_t fake  = (size_t)&((Employee*)0)->age;

    cout << "Real  (base+" << age_offset << ") offset : "
         << real  << "\n";
    cout << "Fake  (0+"    << age_offset << ") offset : "
         << fake  << "\n";
    cout << "Both same?  : "
         << (real == fake ? "YES ✅" : "NO ❌") << "\n";


    return 0;
}
```
