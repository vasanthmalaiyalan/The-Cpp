# புரிந்துகொள்வதற்கு இந்த ஒரே code போதும்! 🎯

---

```cpp
#include <iostream>
using namespace std;

struct Employee
{
    char   id;      
    int    age;     
    double salary;  
};

int main()
{
    // ===================================
    // STEP 1: Cast என்றால் என்ன?
    // ===================================

    int x = 65;
    cout << "(int)   x = " << x          << "\n"; // 65
    cout << "(char)  x = " << (char)x    << "\n"; // A
    cout << "(float) x = " << (float)x   << "\n"; // 65.0
    // value மாறவில்லை — பார்க்கும் முறை மட்டும் மாறியது!


    // ===================================
    // STEP 2: Pointer cast என்றால் என்ன?
    // ===================================

    int    num = 42;
    void*  p1  = (void*)&num;    // address மட்டும்
    int*   p2  = (int*)&num;     // int இருக்கிறது என்று label
    
    cout << "\nvoid* address = " << p1 << "\n";
    cout << "int*  address = " << p2 << "\n";
    // இரண்டும் same address — label மட்டும் வேறு!


    // ===================================
    // STEP 3: (Employee*)0 என்றால் என்ன?
    // ===================================

    Employee* e = (Employee*)0;
    // "0 என்பது Employee இருக்கும் address" — label மட்டும்!
    // எந்த memory-யும் access இல்லை!

    cout << "\nEmployee* e = " << e << "\n"; // 0x0


    // ===================================
    // STEP 4: ->member என்றால் என்ன?
    // ===================================

    // e->age — access செய்தால் crash!
    // ஆனால் address மட்டும் எடுத்தால் safe!

    void* age_addr = (void*)&(e->age);
    cout << "\n&e->age address = " << age_addr << "\n"; // 0x4


    // ===================================
    // STEP 5: offset கணக்கு
    // ===================================

    size_t offset = (size_t)((char*)&((Employee*)0)->age);
    cout << "\noffset of age  = " << offset << "\n"; // 4

    // Standard offsetof:
    cout << "offsetof macro = " << offsetof(Employee, age) << "\n"; // 4


    // ===================================
    // STEP 6: int* ஏன் வேலை செய்யாது?
    // ===================================

    // (int*)0)->age   // ❌ compile error!
    // int-க்கு members இல்லை

    // Employee*-க்கு members இருக்கிறது:
    cout << "\nEmployee layout:\n";
    cout << "id     offset = " << offsetof(Employee, id)     << "\n";
    cout << "age    offset = " << offsetof(Employee, age)    << "\n";
    cout << "salary offset = " << offsetof(Employee, salary) << "\n";

    return 0;
}
```

```
→ offset கணக்கு போட முடிகிறது
→ எந்த memory-யும் தொடவில்லை!
```

---

## நடிப்பு எங்கு நிற்கிறது என்பது முக்கியம்:

```cpp
// நடிப்பு safe:
&((Employee*)0)->age
// "இருப்பதாக நடித்து offset மட்டும் கண்டுபிடி" ✅

// நடிப்பு தாண்டினால்:
*((Employee*)0)
// "உள்ளே போய் value எடு" — 💥 crash!
// உண்மையில் எதுவும் இல்லை என்பது தெரிந்துவிடும்!
```

> **— இதுதான் `offsetof`-இன் முழு ரகசியம்!** 🎯
