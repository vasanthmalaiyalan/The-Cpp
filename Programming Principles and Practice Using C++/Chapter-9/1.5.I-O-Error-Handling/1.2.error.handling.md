### Complete Example

```cpp
#include <fstream>
#include <iostream>
#include <string>
#include <vector>

void fill_vector(
    std::istream& input_stream,
    std::vector<int>& values,
    char terminator)
{
    for (int value {}; input_stream >> value;)
    {
        values.push_back(value);
    }

    if (input_stream.eof())
    {
        std::cout
            << "[INFO] End of file reached\n";

        return;
    }

    if (input_stream.fail())
    {
        input_stream.clear();

        char character {};

        input_stream >> character;

        if (character == terminator)
        {
            std::cout
                << "[INFO] Terminator found: "
                << character
                << '\n';

            return;
        }

        input_stream.unget();

        input_stream.clear(std::ios::failbit);

        throw std::runtime_error(
            "Unexpected character in input");
    }
}

int main()
{
    try
    {
        std::ifstream input_file {"numbers.txt"};

        if (!input_file)
        {
            throw std::runtime_error(
                "Cannot open file");
        }

        input_file.exceptions(
            input_file.exceptions()
            | std::ios::badbit);

        std::vector<int> numbers {};

        fill_vector(
            input_file,
            numbers,
            '*');

        std::cout
            << "\nNumbers read:\n";

        for (int value : numbers)
        {
            std::cout
                << value
                << '\n';
        }

        std::cout
            << "\nFinal stream state:\n";

        std::cout
            << "good(): "
            << input_file.good()
            << '\n';

        std::cout
            << "eof(): "
            << input_file.eof()
            << '\n';

        std::cout
            << "fail(): "
            << input_file.fail()
            << '\n';

        std::cout
            << "bad(): "
            << input_file.bad()
            << '\n';
    }
    catch (const std::exception& exception)
    {
        std::cerr
            << "Error: "
            << exception.what()
            << '\n';

        return 1;
    }
}
```

### Most Important Line in the Whole Chapter

```cpp
for (int value {}; input_stream >> value;)
{
    values.push_back(value);
}
```

This pattern appears everywhere in professional C++:

```cpp
while (file >> value)
while (file >> name >> age)
while (socket >> packet)
while (stream >> token)
```

If you deeply understand **why that loop stops**, **how `fail()` is set**, and **why `clear()` is needed**, then you've understood about 80% of what §9.4 is trying to teach.
