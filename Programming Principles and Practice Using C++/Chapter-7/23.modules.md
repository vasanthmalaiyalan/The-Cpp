# BEST BOOK-ORIENTED MODULE EXAMPLE

# Tokenstream.cppm

```cpp id="6swmc3"
module Tokenstream;

import std;

using namespace std;


// ======================================
// Exported Token class
// ======================================

export class Token
{
public:

    char kind;

    double value;


    Token(char k)
        : kind{k},
          value{0.0}
    {
    }


    Token(char k, double v)
        : kind{k},
          value{v}
    {
    }
};


// ======================================
// Exported Token_stream class
// ======================================

export class Token_stream
{
public:

    Token get();

    void putback(Token t);

private:

    bool full = false;

    Token buffer {'0'};
};


// ======================================
// putback() definition
// ======================================

void Token_stream::putback(Token t)
{
    if (full)
    {
        throw runtime_error(
            "putback() into full buffer"
        );
    }

    buffer = t;

    full = true;
}


// ======================================
// get() definition
// ======================================

Token Token_stream::get()
{
    if (full)
    {
        full = false;

        return buffer;
    }

    return Token {'+'};
}
```
