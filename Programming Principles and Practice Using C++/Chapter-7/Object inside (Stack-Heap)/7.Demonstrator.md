```cpp
import std;


// ==========================================================
// CLASS FOR MEMORY / CAST DEMO
// ==========================================================

class Memory_demo
{
private:

    int id;

    double price;

    char grade;

    std::string name;

public:

    Memory_demo()
    {
        id = 101;

        price = 999.99;

        grade = 'A';

        name = "Vasanth";
    }


    void demonstrate_reinterpret_cast()
    {
        std::cout
            << "\n====================================================\n"
            << "1. reinterpret_cast<char*>\n"
            << "====================================================\n";


        // --------------------------------------------------
        // OBJECT ADDRESS
        // --------------------------------------------------

        std::cout << "this object address           : "
                  << this
                  << '\n';


        // --------------------------------------------------
        // MEMBER ADDRESSES
        // --------------------------------------------------

        std::cout << "&id                           : "
                  << &id
                  << '\n';

        std::cout << "&price                        : "
                  << &price
                  << '\n';

        std::cout << "&grade                        : "
                  << static_cast<void*>(&grade)
                  << '\n';


        // --------------------------------------------------
        // OFFSET CALCULATION
        // --------------------------------------------------

        std::cout
            << "\n---------------- OFFSETS ----------------\n";


        // convert addresses into char*
        // so subtraction becomes BYTE subtraction

        ptrdiff_t id_offset =
            reinterpret_cast<char*>(&id)
          - reinterpret_cast<char*>(this);


        ptrdiff_t price_offset =
            reinterpret_cast<char*>(&price)
          - reinterpret_cast<char*>(this);


        ptrdiff_t grade_offset =
            reinterpret_cast<char*>(&grade)
          - reinterpret_cast<char*>(this);


        std::cout << "id offset                     : "
                  << id_offset
                  << " bytes\n";


        std::cout << "price offset                  : "
                  << price_offset
                  << " bytes\n";


        std::cout << "grade offset                  : "
                  << grade_offset
                  << " bytes\n";


        // --------------------------------------------------
        // RAW MEMORY VIEW
        // --------------------------------------------------

        std::cout
            << "\n---------------- RAW BYTES OF id ----------------\n";


        char* raw_bytes =
            reinterpret_cast<char*>(&id);


        for (size_t i = 0; i < sizeof(id); ++i)
        {
            std::cout
                << "byte[" << i << "] = "
                << static_cast<int>(
                        static_cast<unsigned char>(raw_bytes[i])
                   )
                << '\n';
        }
    }


    void demonstrate_static_cast()
    {
        std::cout
            << "\n====================================================\n"
            << "2. static_cast<const void*>\n"
            << "====================================================\n";


        // --------------------------------------------------
        // WITHOUT CAST
        // --------------------------------------------------

        std::cout
            << "\nWITHOUT static_cast:\n";


        std::cout << "name.data() prints string     : "
                  << name.data()
                  << '\n';


        // --------------------------------------------------
        // WITH CAST
        // --------------------------------------------------

        std::cout
            << "\nWITH static_cast<const void*>:\n";


        std::cout << "actual heap address           : "
                  << static_cast<const void*>(name.data())
                  << '\n';


        // --------------------------------------------------
        // FIRST CHARACTER ADDRESS
        // --------------------------------------------------

        std::cout
            << "\nFIRST CHARACTER:\n";


        std::cout << "*name.data()                  : "
                  << *name.data()
                  << '\n';


        // --------------------------------------------------
        // POINTER ARITHMETIC
        // --------------------------------------------------

        std::cout
            << "\nCHARACTER ADDRESSES:\n";


        for (size_t i = 0; i < name.size(); ++i)
        {
            std::cout
                << "name[" << i << "] = "
                << name[i]
                << " | address = "
                << static_cast<const void*>(name.data() + i)
                << '\n';
        }
    }
};


// ==========================================================
// MAIN
// ==========================================================

int main()
{
    Memory_demo demo;

    demo.demonstrate_reinterpret_cast();

    demo.demonstrate_static_cast();
}
```

# PART 1 — `reinterpret_cast<char*>`

---

# This line

```cpp
reinterpret_cast<char*>(&id)
```

இதன் meaning:

> `id` memory-யை raw bytes மாதிரி பார்க்க.

---

# Why `char*`?

Because:

```text
char = 1 byte
```

அதனால்:
byte-level offset calculation possible.

---

# Offset calculation

```cpp
reinterpret_cast<char*>(&price)
-
reinterpret_cast<char*>(this)
```

இதில்:

* object base address
* member address

இரண்டுக்கும் byte distance கிடைக்கும்.

---

# Raw byte reading

```cpp
char* raw_bytes =
    reinterpret_cast<char*>(&id);
```

இதனால்:
`id` integer memory bytes directly பார்க்கலாம்.

---

# Example output

```text
byte[0] = 101
byte[1] = 0
byte[2] = 0
byte[3] = 0
```

# PART 2 — `static_cast<const void*>`

---

# Without cast

```cpp
std::cout << name.data();
```

Output:

```text
Vasanth
```

ஏன்?

Because:
`char*`
→ string என்று cout treat செய்கிறது.

---

# With cast

```cpp
static_cast<const void*>(name.data())
```

Now output:

```text
0x61ff20
```

Actual heap address.

---

# Why?

Because:
`void*`
means:

```text
generic raw memory address
```

---

# Character addresses

```cpp
name.data() + i
```

இதில்:
ஒவ்வொரு character address print ஆகும்.

---

# Example

```text
V -> 0x9000
a -> 0x9001
s -> 0x9002
```

# Final Big Understanding

## reinterpret_cast

Used for:

* raw memory view
* byte arithmetic
* low-level systems work

---

## static_cast<const void*>

Used for:

* printing actual address
* preventing char* string printing behavior
