```cpp
#include <iostream>
#include <cstdlib>
#include <new>
#include <string>



// ==========================================================
// PERSON CLASS
// ==========================================================

class Person
{
public:

    std::string name;

    Person()
    {
        std::cout
            << "\n[CONSTRUCTOR CALLED]\n";

        name = "Tony Stark";

        std::cout
            << "object address : "
            << this
            << '\n';

        std::cout
            << "name initialized : "
            << name
            << '\n';
    }



    ~Person()
    {
        std::cout
            << "\n[DESTRUCTOR CALLED]\n";

        std::cout
            << "destroying object at : "
            << this
            << '\n';
    }
};



// ==========================================================
// MAIN
// ==========================================================

int main()
{
    std::cout
        << "====================================\n";

    std::cout
        << "STEP 1 : RAW MEMORY ONLY\n";

    std::cout
        << "====================================\n";



    // --------------------------------------------------
    // RAW MEMORY ALLOCATION
    // --------------------------------------------------

    void* raw_memory =
        std::malloc(sizeof(Person));



    std::cout
        << "\nraw memory address : "
        << raw_memory
        << '\n';



    // IMPORTANT:
    // NO CONSTRUCTOR CALLED YET
    // THIS IS NOT A REAL PERSON OBJECT YET



    std::cout
        << "\nNO OBJECT EXISTS YET\n";



    // ==================================================
    // STEP 2
    // CONSTRUCT OBJECT INSIDE MEMORY
    // ==================================================

    std::cout
        << "\n\n====================================\n";

    std::cout
        << "STEP 2 : PLACEMENT NEW\n";

    std::cout
        << "====================================\n";



    Person* person_ptr =
        new(raw_memory) Person();



    std::cout
        << "\nperson_ptr : "
        << person_ptr
        << '\n';



    std::cout
        << "person name : "
        << person_ptr->name
        << '\n';



    // ==================================================
    // STEP 3
    // USE OBJECT
    // ==================================================

    std::cout
        << "\n\n====================================\n";

    std::cout
        << "STEP 3 : USING OBJECT\n";

    std::cout
        << "====================================\n";



    person_ptr->name =
        "Ironman";



    std::cout
        << "\nmodified name : "
        << person_ptr->name
        << '\n';



    // ==================================================
    // STEP 4
    // MANUAL DESTRUCTOR
    // ==================================================

    std::cout
        << "\n\n====================================\n";

    std::cout
        << "STEP 4 : MANUAL DESTRUCTOR\n";

    std::cout
        << "====================================\n";



    person_ptr->~Person();



    std::cout
        << "\nOBJECT DESTROYED\n";



    // IMPORTANT:
    // MEMORY STILL EXISTS
    // ONLY OBJECT LIFETIME ENDED



    // ==================================================
    // STEP 5
    // FREE RAW MEMORY
    // ==================================================

    std::cout
        << "\n\n====================================\n";

    std::cout
        << "STEP 5 : FREE RAW MEMORY\n";

    std::cout
        << "====================================\n";



    std::free(raw_memory);



    std::cout
        << "\nRAW MEMORY FREED\n";
}
```

# இந்த program என்ன கற்பிக்கும்?

## Step 1

```cpp
std::malloc(sizeof(Person))
```

↓

```text
raw bytes மட்டும்
```

🔥 object இல்லை.

---

## Step 2

```cpp
new(raw_memory) Person()
```

↓

```text
constructor runs
real object born
```

🔥

---

## Step 3

```cpp
person_ptr->name
```

↓

இப்போது safe.

ஏனெனில் actual object உருவாகிவிட்டது.

---

## Step 4

```cpp
person_ptr->~Person();
```

↓

```text
object lifetime ends
```

🔥 memory இன்னும் இருக்கும்.

---

## Step 5

```cpp
free(raw_memory);
```

↓

```text
raw storage released
```

🔥

# மிக முக்கியமான concept

இந்த program clearly காட்டுவது:

| Thing               | Separate? |
| ------------------- | --------- |
| raw memory          | yes       |
| object construction | yes       |
| object destruction  | yes       |
| memory release      | yes       |

🔥 இதுதான் advanced C++ memory model.
