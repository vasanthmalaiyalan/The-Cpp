
# delete[] internally roughly does

```text id="q1k7v4"
1. array metadata find
2. destructors call (if needed)
3. memory allocator free()
```

# Very simplified imitation

```cpp id="r9m3x6"
#include <iostream>
#include <cstdlib>



// ==========================================================
// CUSTOM OBJECT
// ==========================================================

class Person
{
public:

    Person()
    {
        std::cout
            << "Person constructor\n";
    }

    ~Person()
    {
        std::cout
            << "Person destructor\n";
    }
};



// ==========================================================
// FAKE DELETE[] IMPLEMENTATION
// ==========================================================

template<typename T>
void fake_delete_array(
    T* ptr,
    size_t count
)
{
    std::cout
        << "\n====================================\n";

    std::cout
        << "FAKE delete[] START\n";

    std::cout
        << "====================================\n";


    // --------------------------------------------------
    // STEP 1
    // CALL DESTRUCTORS
    // --------------------------------------------------

    std::cout
        << "\n[CALLING DESTRUCTORS]\n";


    for (size_t i = 0;
         i < count;
         ++i)
    {
        std::cout
            << "destroying object at : "
            << &ptr[i]
            << '\n';


        // manual destructor call
        ptr[i].~T();
    }



    // --------------------------------------------------
    // STEP 2
    // FREE RAW MEMORY
    // --------------------------------------------------

    std::cout
        << "\n[FREEING HEAP MEMORY]\n";

    std::cout
        << "heap block address : "
        << ptr
        << '\n';


    // raw memory release
    std::free(ptr);


    std::cout
        << "\nMEMORY FREED\n";
}



// ==========================================================
// MAIN
// ==========================================================

int main()
{
    std::cout
        << "\nALLOCATING MEMORY\n";


    // --------------------------------------------------
    // RAW HEAP ALLOCATION
    // --------------------------------------------------

    Person* people =
        (Person*)std::malloc(
            sizeof(Person) * 3
        );


    std::cout
        << "heap address : "
        << people
        << '\n';



    // --------------------------------------------------
    // MANUAL CONSTRUCTION
    // --------------------------------------------------

    std::cout
        << "\n[MANUAL CONSTRUCTION]\n";


    for (int i = 0;
         i < 3;
         ++i)
    {
        new (&people[i]) Person();
    }



    // --------------------------------------------------
    // FAKE delete[]
    // --------------------------------------------------

    fake_delete_array(
        people,
        3
    );
}
```
