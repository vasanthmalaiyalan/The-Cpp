```cpp
#include <iostream>



// ==========================================================
// CUSTOM VECTOR CLASS
// ==========================================================

class My_vector
{
private:

    int* data;

public:

    // --------------------------------------------------
    // CONSTRUCTOR
    // --------------------------------------------------

    My_vector()
    {
        // heap allocation
        data = new int[3];

        data[0] = 10;
        data[1] = 20;
        data[2] = 30;
    }


    // --------------------------------------------------
    // DESTRUCTOR
    // --------------------------------------------------

    ~My_vector()
    {
        delete[] data;
    }


    // --------------------------------------------------
    // OPERATOR []
    // --------------------------------------------------

    int& operator[](size_t i)
    {
        std::cout
            << "\n=====================================\n";

        std::cout
            << "My_vector::operator[] CALLED\n";

        std::cout
            << "=====================================\n";


        std::cout
            << "Requested index        : "
            << i
            << '\n';


        std::cout
            << "data pointer value     : "
            << data
            << '\n';


        std::cout
            << "Calculated address     : "
            << (data + i)
            << '\n';


        std::cout
            << "Returned value         : "
            << *(data + i)
            << '\n';


        return *(data + i);
    }
};



// ==========================================================
// MAIN
// ==========================================================

int main()
{
    // =====================================================
    // 1. NATIVE ARRAY
    // =====================================================

    int arr[3] = {100, 200, 300};


    std::cout
        << "\n\n=====================================================\n";

    std::cout
        << "1. NATIVE ARRAY\n";

    std::cout
        << "=====================================================\n";


    std::cout
        << "\nArray base address:\n";


    std::cout
        << "arr                    : "
        << arr
        << '\n';


    std::cout
        << "\nArray element addresses:\n";


    for (size_t i = 0; i < 3; ++i)
    {
        std::cout
            << "&arr[" << i << "]              : "
            << &arr[i]
            << '\n';
    }


    std::cout
        << "\nUsing arr[2]:\n";


    std::cout
        << "arr[2]                 : "
        << arr[2]
        << '\n';


    std::cout
        << "\nInternal native-array logic:\n";


    std::cout
        << "*(arr + 2)             : "
        << *(arr + 2)
        << '\n';


    std::cout
        << "\nAddress calculation:\n";


    std::cout
        << "arr + 2                : "
        << (arr + 2)
        << '\n';



    // =====================================================
    // 2. CUSTOM VECTOR OBJECT
    // =====================================================

    My_vector numbers;


    std::cout
        << "\n\n=====================================================\n";

    std::cout
        << "2. CUSTOM VECTOR OBJECT\n";

    std::cout
        << "=====================================================\n";


    std::cout
        << "\nObject address:\n";


    std::cout
        << "&numbers               : "
        << &numbers
        << '\n';


    std::cout
        << "\nUsing numbers[2]:\n";


    int value1 = numbers[2];


    std::cout
        << "\nFinal value            : "
        << value1
        << '\n';


    std::cout
        << "\nUsing numbers.operator[](2):\n";


    int value2 = numbers.operator[](2);


    std::cout
        << "\nFinal value            : "
        << value2
        << '\n';
}
```

---

# MOST IMPORTANT DIFFERENCE

---

# 1. Native Array

```cpp
arr[2]
```

NOT function call.

---

# Compiler internally does:

*(arr + 2)

---

# Because array indexing is built into language.

---

# Array memory

```text
arr
0x9000 -> 100
0x9004 -> 200
0x9008 -> 300
```

---

# `arr[2]`

becomes:

0x9000 + (2 \times 4) = 0x9008

---

# Then:

```cpp
*(0x9008)
```

returns:

```text
300
```

---

# NO OBJECT involved.

NO operator overloading.

---

# 2. Custom Vector Object

```cpp
numbers[2]
```

IS function call.

---

# Compiler transforms:

```cpp
numbers.operator[](2)
```

---

# Then:

```cpp
int& operator[](size_t i)
{
    return *(data + i);
}
```

runs.

---

# Therefore

vector indexing:

* object
* function
* heap pointer

involved.

---

# VERY IMPORTANT BIG DIFFERENCE

| Feature                  | Native Array | Vector/Object |
| ------------------------ | ------------ | ------------- |
| `[]` built into language | ✅            | ❌             |
| operator overloading     | ❌            | ✅             |
| object involved          | ❌            | ✅             |
| function call internally | ❌            | ✅             |
| heap usually used        | ❌            | often ✅       |

# Final Most Important Understanding

## Native arrays

`[]`
is:

# language-level indexing syntax.

---

# Objects/vectors

`[]`
is:

# overloaded operator function (`operator[]`).

# One-line summary

> `arr[2]` என்பது built-in pointer arithmetic;
> `numbers[2]` என்பது `numbers.operator[](2)` என்ற function call.
