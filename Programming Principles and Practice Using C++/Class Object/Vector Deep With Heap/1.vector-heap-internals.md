```cpp
#include <iostream>



// ==========================================================
// SIMPLE CUSTOM VECTOR
// ==========================================================

class My_vector
{
private:

    // --------------------------------------------------
    // POINTER TO HEAP MEMORY
    // --------------------------------------------------

    int* data;

    size_t size;

public:

    // --------------------------------------------------
    // CONSTRUCTOR
    // --------------------------------------------------

    My_vector()
    {
        size = 3;

        // heap allocation
        data = new int[size];

        data[0] = 10;
        data[1] = 20;
        data[2] = 30;
    }


    // --------------------------------------------------
    // DESTRUCTOR
    // --------------------------------------------------

    ~My_vector()
    {
        delete[] data;
    }


    // --------------------------------------------------
    // OPERATOR []
    // --------------------------------------------------

    int& operator[](size_t i)
    {
        std::cout
            << "\n=====================================\n";

        std::cout
            << "operator[] CALLED\n";

        std::cout
            << "index requested        : "
            << i
            << '\n';


        // ----------------------------------------------
        // INTERNAL POINTER
        // ----------------------------------------------

        std::cout
            << "data pointer value     : "
            << data
            << '\n';


        // ----------------------------------------------
        // ADDRESS CALCULATION
        // ----------------------------------------------

        std::cout
            << "calculated address     : "
            << (data + i)
            << '\n';


        // ----------------------------------------------
        // VALUE
        // ----------------------------------------------

        std::cout
            << "value at address       : "
            << *(data + i)
            << '\n';


        // ----------------------------------------------
        // RETURN REFERENCE
        // ----------------------------------------------

        return *(data + i);
    }


    // --------------------------------------------------
    // DEBUG FUNCTION
    // --------------------------------------------------

    void print_internal_layout()
    {
        std::cout
            << "\n=====================================\n";

        std::cout
            << "VECTOR OBJECT MEMORY\n";

        std::cout
            << "=====================================\n";


        // vector object address
        std::cout
            << "&vector object         : "
            << this
            << '\n';


        // address of pointer member
        std::cout
            << "&data member           : "
            << &data
            << '\n';


        // actual heap address stored in pointer
        std::cout
            << "data pointer value     : "
            << data
            << '\n';


        // heap elements
        std::cout
            << "\n=====================================\n";

        std::cout
            << "HEAP ELEMENTS\n";

        std::cout
            << "=====================================\n";


        for (size_t i = 0; i < size; ++i)
        {
            std::cout
                << "element[" << i << "] value      : "
                << data[i]
                << '\n';

            std::cout
                << "element[" << i << "] address    : "
                << &data[i]
                << '\n';

            std::cout
                << '\n';
        }
    }
};



// ==========================================================
// MAIN
// ==========================================================

int main()
{
    My_vector numbers;


    // --------------------------------------------------
    // SHOW INTERNAL MEMORY
    // --------------------------------------------------

    numbers.print_internal_layout();


    // --------------------------------------------------
    // NORMAL INDEXING
    // --------------------------------------------------

    std::cout
        << "\n\n=====================================\n";

    std::cout
        << "NORMAL numbers[2]\n";

    std::cout
        << "=====================================\n";


    int value = numbers[2];


    std::cout
        << "\nFINAL RETURNED VALUE : "
        << value
        << '\n';


    // --------------------------------------------------
    // ADDRESS OF ELEMENT
    // --------------------------------------------------

    std::cout
        << "\n\n=====================================\n";

    std::cout
        << "&numbers[2]\n";

    std::cout
        << "=====================================\n";


    std::cout
        << "&numbers[2]           : "
        << &numbers[2]
        << '\n';
}
```

# MOST IMPORTANT THING THIS CODE TEACHES

---

# `numbers`

is:

```text
vector object
```

---

# Inside object

```text
data pointer
```

stored.

---

# Heap memory separate

```text
10
20
30
```

---

# `numbers[2]`

NOT direct language magic.

Actually:

```cpp
numbers.operator[](2)
```

---

# Inside operator[]

```cpp
return *(data + i);
```

---

# Suppose:

```text
data = 0x9000
```

---

# Then:

0x9000 + (2 \times 4) = 0x9008

---

# Then:

```cpp
*(0x9008)
```

returns:

```text
30
```

---

# WHY `&numbers[2]` RETURNS HEAP ADDRESS

Because:

```cpp
operator[] returns int&
```

(reference)

---

# Therefore:

```cpp
&numbers[2]
```

means:

```text
address of referenced heap element
```

---

# VERY IMPORTANT OUTPUT UNDERSTANDING

---

# `&vector object`

```text
stack/object memory
```

---

# `data pointer value`

```text
heap starting address
```

---

# `element[i] address`

```text
actual heap element address
```

---

# FULL MENTAL MODEL

```text
numbers object
+----------------------+
| data ptr -----------+-------> heap
| size                |
+----------------------+


Heap:
0x9000 -> 10
0x9004 -> 20
0x9008 -> 30
```

# `numbers[2]`

internally:

```cpp
*(data + 2)
```

# Final Most Important Understanding

> `numbers` ஒரு object.
> ஆனால் `numbers[i]` என்பது object indexing அல்ல.
>
> அது:
>
> ```cpp
> operator[](i)
> ```
>
> என்ற function call.
>
> அந்த function:
>
> ```cpp

> ```
